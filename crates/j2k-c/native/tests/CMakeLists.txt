cmake_minimum_required(VERSION 3.20)
project(j2k_consumer_tests LANGUAGES C CXX)
find_package(J2K CONFIG REQUIRED)
if(NOT EXISTS "${J2K_FIXTURE}")
    message(FATAL_ERROR "Set J2K_FIXTURE to the generated make_fixture image")
endif()

enable_testing()
add_executable(c_consumer consumer.c)
target_compile_features(c_consumer PRIVATE c_std_11)
target_link_libraries(c_consumer PRIVATE J2K::j2k)
add_executable(cpp_consumer consumer.cpp)
target_compile_features(cpp_consumer PRIVATE cxx_std_17)
target_link_libraries(cpp_consumer PRIVATE J2K::j2k)
foreach(consumer c_consumer cpp_consumer)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${consumer} PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME ${consumer} COMMAND ${consumer} "${J2K_FIXTURE}")
endforeach()
